<script>
// <interpretar>
// Entiende frases en castellano o catalán como «examen de fluidos el jueves a las 10»
// o «sopar amb la Marta demà a les 21h». hoy: 'yyyy-mm-dd'.
// Devuelve { titulo, dia, diaFin, hora, fin, min, todoDia, tipo, tarea, lugar }: dia y hora quedan vacíos si la frase no los dice.
// Este bloque está copiado igual en Codigo.gs y en Panel.html (node sincronizar.js lo copia desde interpretar.js).
function interpretar(texto, hoy) {
  const orig = String(texto || '').replace(/[’`´]/g, "'").replace(/\s+/g, ' ').trim();
  // Copia en minúsculas y sin acentos con la misma longitud: así las posiciones sirven para recortar el título.
  const n = orig.split('').map(c => {
    const s = c.toLowerCase().normalize('NFD').replace(/[̀-ͯ]/g, '');
    return s.length === 1 ? s : c;
  }).join('');
  const usado = new Array(n.length).fill(false);
  // Primera coincidencia que no pisa nada ya reconocido; la marca como usada.
  const tomar = re => {
    re.lastIndex = 0;
    let m;
    while ((m = re.exec(n))) {
      let libre = true;
      for (let i = m.index; i < m.index + m[0].length; i++) if (usado[i]) libre = false;
      if (libre && m[0]) {
        for (let i = m.index; i < m.index + m[0].length; i++) usado[i] = true;
        return m;
      }
      if (!m[0]) re.lastIndex++;
    }
    return null;
  };

  const r = { titulo: '', dia: '', diaFin: '', hora: '', fin: '', min: 0, todoDia: false, tipo: 'evento', tarea: false, lugar: '' };
  const [y0, m0, d0] = String(hoy).split('-').map(Number);
  const base = Date.UTC(y0, m0 - 1, d0);
  const iso = t => new Date(t).toISOString().slice(0, 10);
  const mas = k => iso(base + k * 86400000);
  const dow = (new Date(base).getUTCDay() + 6) % 7; // 0 = lunes
  const dosCifras = x => String(x).padStart(2, '0');
  const fecha = (y, m, d) => {
    const t = Date.UTC(y, m - 1, d);
    const f = new Date(t);
    return f.getUTCMonth() === m - 1 && f.getUTCDate() === d ? iso(t) : '';
  };
  // Día y mes sin año: el próximo que llegue (si ya pasó este año, el año que viene).
  const proxima = (m, d) => {
    const f = fecha(y0, m, d);
    return f && f < hoy ? fecha(y0 + 1, m, d) : f;
  };

  // Tarea en vez de evento.
  if (tomar(/^(?:tareas?|tascas?|pendiente|recordar|recorda'm|recordatorio)\b[\s:,.-]*/g)) r.tarea = true;

  // Lugar al final: «… @ biblioteca».
  const lugar = tomar(/\s*@\s*(.+)$/g);
  if (lugar) r.lugar = orig.substr(lugar.index + lugar[0].indexOf('@') + 1).trim();

  if (tomar(/\b(?:todo el dia|tot el dia)\b/g)) r.todoDia = true;

  // Franja del día: sirve de hora si no se dice otra y para saber si «a las 5» es por la tarde.
  let franja = '';
  const fr = tomar(/\b(?:por la|de la|a la|per la|pel|al|esta|aquesta)\s+(manana|mati|tarde|tarda|noche|nit|mediodia|migdia)\b/g);
  if (fr) {
    franja = { manana: 'manana', mati: 'manana', tarde: 'tarde', tarda: 'tarde', noche: 'noche', nit: 'noche', mediodia: 'mediodia', migdia: 'mediodia' }[fr[1]];
    if (/^(esta|aquesta)\b/.test(fr[0])) r.dia = mas(0);
  }

  // Día.
  const rel = r.dia ? null : tomar(/\b(pasado manana|passat dema|dema passat|hoy|avui|manana|dema)\b/g);
  if (rel) {
    const w = rel[1];
    r.dia = /pasado|passat/.test(w) ? mas(2) : /manana|dema/.test(w) ? mas(1) : mas(0);
  }
  if (!r.dia) {
    const en = tomar(/\b(?:en|dentro de|d'aqui a|d'aqui|daqui a)\s+(\d{1,3})\s+(dias?|dies|semanas?|setmanes?)\b/g);
    if (en) r.dia = mas(Number(en[1]) * (/^s/.test(en[2]) ? 7 : 1));
  }
  const DIAS = ['lunes|dilluns', 'martes|dimarts', 'miercoles|dimecres', 'jueves|dijous', 'viernes|divendres', 'sabado|dissabte', 'domingo|diumenge'];
  if (!r.dia) {
    const sem = tomar(/\b(?:(?:el|este|aquest|el proximo|el proxim|proximo|proxim)\s+)?(lunes|martes|miercoles|jueves|viernes|sabado|domingo|dilluns|dimarts|dimecres|dijous|divendres|dissabte|diumenge)(?:\s+(que viene|que ve|vinent|proximo|proxim|de la semana que viene|de la setmana que ve))?\b/g);
    if (sem) {
      const idx = DIAS.findIndex(d => d.split('|').indexOf(sem[1]) >= 0);
      // «El jueves» dicho un jueves es el de la semana que viene.
      let delta = (idx - dow + 7) % 7 || 7;
      if (sem[2] && /semana|setmana/.test(sem[2]) && delta < 7 && idx >= dow) delta += 7;
      r.dia = mas(delta);
    }
  }
  const MESES = ['ene|gen', 'feb', 'mar', 'abr', 'may|mai', 'jun', 'jul', 'ago', 'sep|set', 'oct', 'nov', 'dic|des'];
  const numMes = w => MESES.findIndex(alts => alts.split('|').some(a => w.indexOf(a) === 0)) + 1;
  const MES_RE = '(enero|gener|febrero|febrer|marzo|marc|abril|mayo|maig|junio|juny|julio|juliol|agosto|agost|septiembre|setiembre|setembre|octubre|noviembre|novembre|diciembre|desembre|ene|feb|mar|abr|jun|jul|ago|sept|sep|set|oct|nov|dic|des)';
  // Varios días seguidos: «del 6 al 9 de diciembre» → todo el día, de ida a vuelta.
  if (!r.dia) {
    const dd = tomar(new RegExp("\\b(?:del|de|desde el|des del)\\s+(\\d{1,2})\\s+(?:al|a|hasta el|fins al)\\s+(\\d{1,2})\\s*(?:de\\s+|d'|del\\s+)?" + MES_RE + '\\b', 'g'));
    if (dd && Number(dd[1]) <= Number(dd[2])) {
      r.dia = proxima(numMes(dd[3]), Number(dd[1]));
      r.diaFin = r.dia ? fecha(Number(r.dia.slice(0, 4)), numMes(dd[3]), Number(dd[2])) : '';
      r.todoDia = !!r.diaFin;
    }
  }
  if (!r.dia) {
    const dm = tomar(/\b(?:el\s+)?(?:dia\s+)?(\d{1,2})\s*(?:de\s+|d'|del\s+)?(enero|gener|febrero|febrer|marzo|marc|abril|mayo|maig|junio|juny|julio|juliol|agosto|agost|septiembre|setiembre|setembre|octubre|noviembre|novembre|diciembre|desembre|ene|feb|mar|abr|jun|jul|ago|sept|sep|set|oct|nov|dic|des)\b\.?(?:\s+(?:de\s+|del\s+)?(\d{4}))?/g);
    if (dm) r.dia = dm[3] ? fecha(Number(dm[3]), numMes(dm[2]), Number(dm[1])) : proxima(numMes(dm[2]), Number(dm[1]));
  }

  // Horas: «10:30», «10.30», «10h», «10h30» o un número tras «a las».
  const NUM = '(\\d{1,2})(?:[:.](\\d{2})|h(\\d{2})?)?';
  const ART = "(?:(?:las?|les|l')\\s*)?";
  // Entrenar se hace temprano: «correr a las 6» son las 6 de la mañana.
  const madruga = /\b(entreno|entrenar|entrenament|correr|calistenia|movilidad|mobilitat|gimnasio|gym|running|rodaje)\b/.test(n);
  const minutos = (h, m, crudo) => {
    h = Number(h);
    m = Number(m || 0);
    if (h > 24 || m > 59) return null;
    // Sin «de la mañana», de 1 a 7 es por la tarde («a las 5» → 17:00), salvo que se escriba con cero delante.
    if (franja === 'manana' || /^0/.test(crudo)) return h * 60 + m;
    if ((franja === 'tarde' || franja === 'noche') && h < 12) h += 12;
    else if (!franja && h >= 1 && h <= 7 && !(madruga && h >= 5)) h += 12;
    return (h % 24) * 60 + m;
  };
  let ini = null;
  let fin = null;
  const rango = tomar(new RegExp('\\b(?:de|desde|des de|entre)\\s+' + ART + NUM + '\\s*(?:a|y|i|hasta|fins a|fins)\\s+' + ART + NUM + '(?:\\s*(?:h|horas|hores)\\b)?', 'g'))
    || tomar(new RegExp('\\b(\\d{1,2})(?:[:.](\\d{2})|h(\\d{2})?)\\s*-\\s*' + NUM + '\\b', 'g'));
  if (rango) {
    ini = minutos(rango[1], rango[2] || rango[3], rango[1]);
    fin = minutos(rango[4], rango[5] || rango[6], rango[4]);
    if (ini != null && fin != null && fin <= ini && fin + 720 > ini) fin += 720;
  }
  if (ini == null) {
    const t = tomar(new RegExp('\\b(?:a|sobre|cap a|hacia|at)\\s+' + ART + NUM + '(?:\\s*(?:h|horas|hores)\\b)?', 'g'));
    if (t) ini = minutos(t[1], t[2] || t[3], t[1]);
  }
  // Duración: «2 horas», «durante 1h30», «45 min».
  let dur = 0;
  const d1 = tomar(/\b(?:durante|durant|por|per)\s+(\d+(?:[.,]\d+)?)\s*(?:h(\d{2})?|horas?|hores?|min|minutos?|minuts?)\b/g)

  if (d1) {
    const v = Number(d1[1].replace(',', '.'));
    dur = /min/.test(d1[0]) ? v : v * 60 + Number(d1[2] || 0);
  }
  if (ini == null) {
    // «10h», «10:30» o «10.30» sueltos. Un «2h» suelto (de 1 a 5) se entiende como duración.
    const t = tomar(/\b(\d{1,2})(?:[:.](\d{2})|h(\d{2})?)(?![\d/-])/g);
    if (t) {
      if (!t[2] && !t[3] && Number(t[1]) <= 5 && !dur) dur = Number(t[1]) * 60;
      else ini = minutos(t[1], t[2] || t[3], t[1]);
    }
  }
  if (!dur) {
    const d2 = tomar(/\b(\d)h(\d{2})\b/g);
    if (d2) dur = Number(d2[1]) * 60 + Number(d2[2]);
  }

  // Fechas con números, después de las horas para no confundir «10.30» con una fecha.
  if (!r.dia) {
    const num = tomar(/\b(?:el\s+)?(\d{1,2})[/-](\d{1,2})(?:[/-](\d{2,4}))?\b/g);
    if (num) {
      const y = num[3] ? Number(num[3].length === 2 ? '20' + num[3] : num[3]) : 0;
      r.dia = y ? fecha(y, Number(num[2]), Number(num[1])) : proxima(Number(num[2]), Number(num[1]));
    }
  }
  if (!r.dia) {
    const sol = tomar(/\b(?:el|dia|el dia)\s+(\d{1,2})\b(?!\s*(?:h\b|:|horas|hores|min))/g);
    if (sol) {
      const d = Number(sol[1]);
      let f = fecha(y0, m0, d);
      if (!f || f < hoy) f = m0 === 12 ? fecha(y0 + 1, 1, d) : fecha(y0, m0 + 1, d);
      r.dia = f;
    }
  }

  if (ini == null && franja && !r.todoDia) ini = { manana: 540, mediodia: 840, tarde: 1020, noche: 1260 }[franja];

  // Tipo por palabras clave (el orden importa: «entregar práctica» es una entrega, no una clase).
  const TIPOS = [
    ['estudio', /\b(estudiar|estudio|estudi|repasar|repaso|repassar|repas|sesion de estudio)\b/],
    ['viaje', /\b(viaje|viatge|escapada|salida|sortida|excursion|excursio|acampada|campaments?|colonias|colonies)\b/],
    ['entreno', /\b(entreno|entrenar|entrenament|gimnasio|gym|correr|corriendo|calistenia|movilidad|mobilitat|running|rodaje|series)\b/],
    ['examen', /\b(examen|examenes|examens|exam|parcial|parcials|final de|prueba|control|test|recuperacion|recuperacio)\b/],
    ['entrega', /\b(entrega|entregas|entregar|lliurament|lliurar|deadline|fecha limite|data limit)\b/],
    ['clase', /\b(clase|clases|classe|classes|teoria|practicas?|laboratorio|lab|seminario|seminari|tutoria)\b/],
    ['cita', /\b(medico|metge|dentista|cita|consulta|fisio|peluqueria|perruqueria|barbero|revision|revisio|banco|banc|entrevista)\b/],
    ['plan', /\b(cena|sopar|comida|dinar|quedada|quedar|fiesta|festa|cumple|cumpleanos|aniversari|cine|cinema|concierto|concert|partido|partit|birras?|cafe)\b/],
  ];
  const tipo = TIPOS.find(t => t[1].test(n));
  if (tipo) r.tipo = tipo[0];

  if (ini != null) {
    r.hora = dosCifras(Math.floor(ini / 60)) + ':' + dosCifras(ini % 60);
    r.min = fin != null ? fin - ini : dur || ({ examen: 120, plan: 120, estudio: 50 }[r.tipo] || 60);
    const f = ini + r.min;
    r.fin = dosCifras(Math.floor(f / 60) % 24) + ':' + dosCifras(f % 60);
  } else {
    r.min = dur;
    if (!r.tarea) r.todoDia = true;
  }

  // Título: lo que queda sin reconocer, sin palabras sueltas en los bordes.
  let t = orig.split('').map((c, i) => usado[i] ? ' ' : c).join('').replace(/\s+/g, ' ').trim();
  const SUELTAS = "el|la|los|las|lo|de|del|a|al|en|y|i|e|amb|con|para|per|que|les|l'|d'|por|este|esta|aquest|aquesta|sobre|hacia|cap|durante|durant|desde|des|hasta|fins";
  const borde = new RegExp('^(?:(?:' + SUELTAS + ")(?:\\s+|$)|[,.;:\\-–])+|(?:\\s+(?:" + SUELTAS + ')|\\s*[,.;:\\-–])+$', 'i');
  for (let i = 0; i < 4; i++) t = t.replace(borde, '').trim();
  r.titulo = t ? t.charAt(0).toUpperCase() + t.slice(1) : '';
  return r;
}

    html += `<form class="rapido" data-form="rapido"><input id="rapido" placeholder="Apunta algo: «estudiar STE el lunes a las 4»" aria-label="Apuntar un evento" autocomplete="off" enterkeyhint="send"><button aria-label="Apuntar"><svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.4" stroke-linecap="round"><path d="M12 5v14M5 12h14"/></svg></button></form>
      <p class="previa" id="previa"></p>`;
    return html;
  }

  function filaEvento(e, pasado) {
    const sub = e.tipo === 'estudio' ? primeraLinea(e.desc) : e.lugar;
    return `<li><button class="${pasado ? 'pasado' : ''} ${e.tipo === 'estudio' ? 'estudio' : ''}" data-act="ver" data-ev="${esc(clave(e))}">
        <time>${e.hora}</time><i style="background:${COLOR[e.tipo] || COLOR.evento}"></i>
        <span style="min-width:0"><b>${esc(e.titulo)}</b>${sub ? `<small>${esc(sub)}</small>` : ''}</span>
        <span class="ok">${e.estado === 'hecha' ? OK : ''}</span></button></li>`;
  }

  function vistaExamenes() {
    if (!D.examenes.length) {
      return `<p class="vacio">No hay exámenes ni entregas a la vista.</p>
        <button class="btn ancho sec" data-act="nuevo" data-modo="examen">+ Examen o entrega</button>`;
    }
    return `<div class="examenes">${D.examenes.map(e => {
      const k = e.id + '|' + e.t;
      const color = COLOR[e.tipo];
      const sin = e.tipo === 'examen' && e.asig && !e.minTotal;
      return `<button class="ex" data-act="abrir" data-k="${esc(k)}">
          <div class="n" style="color:${color}">${e.faltan}<small>${e.faltan === 1 ? 'día' : 'días'}</small></div>
          <b>${e.asig ? `<span class="asig">${esc(e.asig)}</span>` : ''}${esc(sinAsig(e))}</b>
          <span class="cuando">${fecha(e.dia)}${e.hora ? ' · ' + e.hora : ''}${e.lugar ? ' · ' + esc(e.lugar) : ''}</span>
          ${e.minTotal ? `<div class="barra"><span><em style="width:${Math.min(100, e.minHechos / e.minTotal * 100)}%"></em></span>${horas(e.minHechos)} de ${horas(e.minTotal)}</div>` : ''}
          ${e.prox ? `<div class="prox">Próxima: ${fecha(e.prox.dia)} · ${e.prox.hora}</div>` : sin ? `<div class="prox falta">Sin sesiones de estudio todavía</div>` : ''}
          ${abierto === k && e.sesiones.length ? `<ul class="sesiones">${e.sesiones.map(s => `<li>
              <span class="${s.hecha ? 'si' : s.pasada ? 'no' : ''}">${s.hecha ? '✓' : s.pasada ? '!' : '·'}</span>
              <span>${fecha(s.dia)} ${s.hora}</span><em>${esc(sinAsig({ asig: e.asig, titulo: s.titulo }))}</em></li>`).join('')}</ul>` : ''}
        </button>`;
    }).join('')}</div>
      <button class="btn ancho sec" data-act="nuevo" data-modo="examen">+ Examen o entrega</button>`;
  }

  function filaTarea(t, hoy) {
    let nota = '';
    if (!t.hecha && t.dia) nota = t.dia < hoy ? `<small class="vencida">Era para el ${fecha(t.dia)}</small>` : t.dia === hoy ? '' : `<small>${fecha(t.dia)}</small>`;
    return `<div class="tarea ${t.hecha ? 'ok' : ''}"><button data-act="tarea" data-id="${esc(t.id)}"><span class="c">${t.hecha ? OK : ''}</span><span><b>${esc(t.titulo)}</b>${nota}</span></button>
      <button class="quitar" data-act="borrarTarea" data-id="${esc(t.id)}" aria-label="Borrar tarea">${X}</button></div>`;
  }

  function vistaTareas(hoy) {
    const grupos = [
      ['Atrasadas', t => !t.hecha && t.dia && t.dia < hoy],
      ['Hoy', t => !t.hecha && t.dia === hoy],
      ['Próximas', t => !t.hecha && t.dia > hoy],
      ['Sin fecha', t => !t.hecha && !t.dia],
      ['Hechas', t => t.hecha],
    ];
    let html = `<form class="rapido" data-form="tarea"><input id="nuevaTarea" placeholder="Nueva tarea: «imprimir apuntes el lunes»" aria-label="Nueva tarea" autocomplete="off" enterkeyhint="done"><button aria-label="Añadir tarea"><svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.4" stroke-linecap="round"><path d="M12 5v14M5 12h14"/></svg></button></form>
      <p class="previa" id="previaTarea"></p>`;
    let alguna = false;
    grupos.forEach(([nombre, f]) => {
      const lista = D.tareas.filter(f);
      if (!lista.length) return;
      alguna = true;
      html += `<section class="grupo"><span class="eti">${nombre}</span>${lista.map(t => filaTarea(t, hoy)).join('')}</section>`;
    });
    if (!alguna) html += '<p class="vacio">Nada pendiente.</p>';
    return html;
  }

  // Lo que se va a apuntar, mientras se escribe.
  function previa(texto, donde, comoTarea) {
    if (!donde) return;
    if (!texto.trim()) { donde.innerHTML = ''; return; }
    const r = interpretar(texto, hoyLocal());
    if (!r.titulo) { donde.innerHTML = ''; return; }
    if (comoTarea || r.tarea) {
      donde.innerHTML = `Tarea <b>${esc(r.titulo)}</b>${r.dia ? ' · ' + fecha(r.dia) : ''}`;
      return;
    }
    const dia = r.dia || hoyLocal();
    const cuando = r.diaFin ? fecha(dia) + ' – ' + fecha(r.diaFin) : r.hora ? fecha(dia) + ' · ' + r.hora + '–' + r.fin : fecha(dia) + ' · todo el día';
    donde.innerHTML = `${NOMBRE[r.tipo] || 'Evento'} <b>${esc(r.titulo)}</b> · ${cuando}`;
  }

  // ---------- Hojas ----------

  function abrirHoja(html) {
    $('velo').innerHTML = `<div class="hoja" role="dialog" aria-modal="true">${html}</div>`;
    $('velo').hidden = false;
  }
  function cerrarHoja() { $('velo').hidden = true; $('velo').innerHTML = ''; }

  function verEvento(e) {
    const cuando = e.todoDia
      ? fecha(e.dia) + (e.diaFin > e.dia ? ' – ' + fecha(e.diaFin) : ' · todo el día')
      : fecha(e.dia) + ' · ' + e.hora + '–' + e.horaFin;
    const estudio = e.tipo === 'estudio' && !e.todoDia;
    abrirHoja(`<span class="eti"><i style="display:inline-block;width:8px;height:8px;border-radius:4px;background:${COLOR[e.tipo] || COLOR.evento};margin-right:6px"></i>${NOMBRE[e.tipo] || 'Evento'}${e.asig ? ' · ' + esc(e.asig) : ''}</span>
      <h3>${esc(e.titulo)}</h3>
      <p class="cuando">${cuando}${e.lugar ? ' · ' + esc(e.lugar) : ''}</p>
      ${e.desc ? `<p class="como">${esc(e.desc)}</p>` : ''}
      ${estudio ? `<div class="fila">
          <button class="btn" style="flex:1" data-act="${e.estado === 'hecha' ? 'deshacer' : 'hecha'}" data-ev="${esc(clave(e))}">${e.estado === 'hecha' ? 'Marcar como pendiente' : 'Hecha'}</button>
          <button class="btn sec" style="flex:1" data-act="recolocar" data-ev="${esc(clave(e))}">Muévela</button></div>` : ''}
      ${e.editable ? `<div class="fila">
          <button class="btn peligro" style="flex:1" data-act="borrar" data-ev="${esc(clave(e))}">${e.serie ? 'Borrar solo este' : 'Borrar'}</button>
          ${e.serie ? `<button class="btn peligro" style="flex:1" data-act="borrar" data-serie="1" data-ev="${esc(clave(e))}">Borrar toda la serie</button>` : ''}</div>` : ''}
      <button class="btn sec ancho" data-act="cerrar">Cerrar</button>`);
  }

  let modo = 'evento';
  let tipoMeta = 'examen';
  function nuevo(m) {
    modo = m || modo;
    const hoy = hoyLocal();
    const asigs = [...new Set(D.examenes.map(e => e.asig).concat(D.eventos.map(e => e.asig)).filter(Boolean))];
    const pestañas = [['evento', 'Evento'], ['viaje', 'Viaje'], ['examen', 'Examen'], ['tarea', 'Tarea']];
    let cuerpo = '';
    if (modo === 'evento') {
      cuerpo = `<input class="campo" id="nTexto" placeholder="«estudiar STE el sábado a las 5»" autocomplete="off">
        <p class="previa" id="nPrevia" style="margin:0"></p>
        <p class="nota">Escribe o dicta con el micro del teclado. Entiende el día, la hora, «@ lugar» y el tipo (estudiar, examen, entreno, cita…).</p>`;
    } else if (modo === 'viaje') {
      cuerpo = `<input class="campo" id="nTitulo" placeholder="Florencia, sortida del cau…" autocomplete="off">
        <div class="dos"><label>Ida<input class="campo" type="date" id="nDia" value="${hoy}"></label><label>Vuelta<input class="campo" type="date" id="nFin" value="${hoy}"></label></div>
        <p class="nota">Esos días no se te recolocará ninguna sesión de estudio.</p>`;
    } else if (modo === 'examen') {
      cuerpo = `<div class="chips">${[['examen', 'Examen'], ['entrega', 'Entrega']].map(([v, t]) => `<button type="button" data-act="tipoMeta" data-v="${v}" aria-pressed="${tipoMeta === v}">${t}</button>`).join('')}</div>
        <input class="campo" id="nTitulo" placeholder="${tipoMeta === 'examen' ? '1er control, final…' : 'Informe práctica 3…'}" autocomplete="off">
        <input class="campo" id="nAsig" placeholder="Asignatura (STE, MF, STM…)" list="asigs" autocomplete="off" autocapitalize="characters">
        <datalist id="asigs">${asigs.map(a => `<option value="${esc(a)}">`).join('')}</datalist>
        <div class="dos"><label>Día<input class="campo" type="date" id="nDia" value="${hoy}"></label><label>Hora (si la sabes)<input class="campo" type="time" id="nHora"></label></div>
        <p class="nota">Con la asignatura, Plum cuenta las horas de estudio que llevas para ${tipoMeta === 'examen' ? 'este examen' : 'esta entrega'}.</p>`;
    } else {
      cuerpo = `<input class="campo" id="nTexto" placeholder="«imprimir apuntes el lunes»" autocomplete="off">
        <p class="previa" id="nPrevia" style="margin:0"></p>`;
    }
    abrirHoja(`<form data-form="nuevo" style="display:contents">
      <h3>Añadir</h3>
      <div class="chips">${pestañas.map(([v, t]) => `<button type="button" data-act="modo" data-v="${v}" aria-pressed="${modo === v}">${t}</button>`).join('')}</div>
      ${cuerpo}
      <div class="fila"><button class="btn" style="flex:1">Guardar</button><button type="button" class="btn sec" data-act="cerrar">Cancelar</button></div>
    </form>`);
    const t = $('nTexto');
    if (t) t.oninput = () => previa(t.value, $('nPrevia'), modo === 'tarea');
    const foco = $('nTexto') || $('nTitulo');
    if (foco) setTimeout(() => foco.focus(), 60);
  }

  async function guardarNuevo(boton) {
    if (modo === 'evento' || modo === 'tarea') {
      const texto = $('nTexto').value.trim();
      if (!texto) return avisar('Escribe qué apuntar');
      const r = interpretar(texto, hoyLocal());
      if (modo === 'evento' && r.diaFin) {
        return hacer('evento', { titulo: r.titulo, tipo: r.tipo, dia: r.dia, diaFin: r.diaFin, todoDia: true, lugar: r.lugar }, boton);
      }
      return hacer('rapido', { texto, tarea: modo === 'tarea' }, boton);
    }
    const titulo = $('nTitulo').value.trim();
    if (!titulo) return avisar('Ponle un nombre');
    if (modo === 'viaje') {
      const dia = $('nDia').value, diaFin = $('nFin').value || dia;
      if (diaFin < dia) return avisar('La vuelta es antes de la ida');
      return hacer('evento', { titulo, tipo: 'viaje', dia, diaFin, todoDia: true }, boton);
    }
    const hora = $('nHora').value;
    return hacer('evento', { titulo, tipo: tipoMeta, asig: $('nAsig').value.trim().toUpperCase(), dia: $('nDia').value, hora, todoDia: !hora, min: tipoMeta === 'examen' ? 120 : 30 }, boton);
  }

  // ---------- Toques ----------

  document.addEventListener('click', e => {
    if (e.target === $('velo')) return cerrarHoja();
    const b = e.target.closest('[data-tab],[data-act]');
    if (!b) return;
    if (b.dataset.tab) {
      tab = b.dataset.tab;
      try { localStorage.setItem('plum.tab', tab); } catch (err) {}
      $('vista').scrollTop = 0;
      return pintar();
    }
    const a = b.dataset.act;
    const ev = b.dataset.ev ? buscar(b.dataset.ev) : null;
    if (a === 'reintentar') return cargar();
    if (a === 'cerrar') return cerrarHoja();
    if (a === 'ver' && ev) return verEvento(ev);
    if (a === 'hecha' && ev) return hacer('estado', Object.assign(ref(ev), { estado: 'hecha' }), b);
    if (a === 'deshacer' && ev) return hacer('estado', Object.assign(ref(ev), { estado: '' }), b);
    if (a === 'recolocar' && ev) return hacer('recolocar', ref(ev), b);
    if (a === 'borrar' && ev) {
      // Dos toques para borrar: el primero avisa.
      if (!b.classList.contains('seguro')) {
        b.classList.add('seguro');
        b.textContent = 'Toca otra vez para borrar';
        return;
      }
      return hacer('borrar', Object.assign(ref(ev), { serie: !!b.dataset.serie }), b);
    }
    if (a === 'abrir') { abierto = abierto === b.dataset.k ? '' : b.dataset.k; return pintar(); }
    if (a === 'nuevo') return nuevo(b.dataset.modo);
    if (a === 'modo') return nuevo(b.dataset.v);
    if (a === 'tipoMeta') { tipoMeta = b.dataset.v; return nuevo('examen'); }
    if (a === 'tarea') return hacer('hecha', { id: b.dataset.id }, b);
    if (a === 'borrarTarea') return hacer('borrarTarea', { id: b.dataset.id }, b);
  });

  $('mas').onclick = () => nuevo(tab === 'tasks' ? 'tarea' : tab === 'exams' ? 'examen' : 'evento');

  document.addEventListener('submit', e => {
    e.preventDefault();
    const f = e.target.dataset.form;
    const boton = e.target.querySelector('button:not([type=button])');
    if (f === 'rapido') {
      const texto = $('rapido').value.trim();
      if (texto) hacer('rapido', { texto }, boton);
    }
    if (f === 'tarea') {
      const texto = $('nuevaTarea').value.trim();
      if (texto) hacer('rapido', { texto, tarea: true }, boton);
    }
    if (f === 'nuevo') guardarNuevo(boton);
  });

  // Al volver a la app después de un rato, se refresca sola.
  document.addEventListener('visibilitychange', () => {
    if (document.visibilityState === 'visible' && Date.now() - cargadoEn > 60000) cargar();
  });
  setInterval(() => { if (D && document.visibilityState === 'visible' && tab === 'today' && $('velo').hidden && document.activeElement.tagName !== 'INPUT') pintar(); }, 60000);

  // Arranque: lo último guardado al momento y, detrás, lo de verdad.
  if (!EN_GOOGLE && (!API || !K)) {
    pantallaClave();
  } else {
    const cache = leerCache();
    if (cache) { D = cache; pintar(); }
    cargar();
  }
</script>
